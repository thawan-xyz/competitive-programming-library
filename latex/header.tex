\newcommand{\thumbtab}[1]{%
  \setlength{\tabheight}{\dimexpr(\paperheight - 1.2cm) / \numsections\relax}%
  % the name shrinks if it is longer than the tab
  \sbox{\tabname}{\color{white}\fontsize{5pt}{5.5pt}\selectfont\bfseries\csname secname#1\endcsname}%
  \ifdim\wd\tabname>\dimexpr\tabheight - 6pt\relax
    \sbox{\tabname}{\resizebox{\dimexpr\tabheight - 6pt\relax}{!}{\usebox{\tabname}}}%
  \fi
  % drawn from the upper left corner of the page, at shipout
  \put(0, 0){\begin{tikzpicture}[overlay]
    \coordinate (top) at (\paperwidth, {-0.6cm - (#1 - 1) * \tabheight});
    \fill[sec#1] (top) rectangle ++(-\tabwidth, -\tabheight);
    \node[rotate=-90, inner sep=0pt] at ($(top) + (-0.5 * \tabwidth, -0.5 * \tabheight)$) {\usebox{\tabname}};
  \end{tikzpicture}}%
}

% sections of the page: from the one open at its top to the last one that starts on it
\newcount\firstsec
\newcount\lastsec
\newcommand{\findsections}{%
  \global\firstsec=0 \global\lastsec=0
  \foreach \k in {1,...,\numsections} {%
    \ifnum\getpagerefnumber{s\k}<\value{page}\global\firstsec=\k\fi
    \ifnum\getpagerefnumber{s\k}>\value{page}\else\global\lastsec=\k\fi
  }%
  \ifnum\firstsec=0 \global\firstsec=\lastsec\fi
  \ifnum\firstsec=0 \else\ifnum\getpagerefnumber{s\the\firstsec}<\value{page}\else
    \foreach \k in {1,...,\numsections} {\ifnum\getpagerefnumber{s\k}=\value{page}\ifnum\k<\firstsec\global\firstsec=\k\fi\fi}%
  \fi\fi
}
\newcommand{\sectiontag}[1]{{\setlength{\fboxsep}{1.2pt}\colorbox{sec#1}{\color{white}\scriptsize\bfseries\vphantom{dg}\,\csname secname#1\endcsname\,}}}

\lhead{\findsections\ifnum\lastsec>0 \foreach \k in {\the\firstsec,...,\the\lastsec} {\sectiontag{\k}\ }\fi}
