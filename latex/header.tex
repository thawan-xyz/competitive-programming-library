\newcommand{\secbar}[2]{%

  \pdfbookmark[0]{#2}{sec#1}%
  \barbox{black!85}{\color{white}\fontsize{7pt}{7.6pt}\selectfont\bfseries}{#2}%
  \vskip 0.8pt\nobreak
}

\newcommand{\filebar}[3]{%
  \par\addvspace{1.5pt}%
  \phantomsection\label{f#1}\pdfbookmark[1]{#2}{file#1}%

  \barbox{gray!22}{\fontsize{7pt}{7.6pt}\selectfont\bfseries}%
    {#2\hfill{\mdseries\tiny\color{black!60}#3}}%
  \vskip 0.8pt\nobreak
}

\newcommand{\toc@place}{%
  \ifnum\toccol>\col@number \toccap=\dimexpr\@colht-1pt\relax \else \toccap=\toc@firstht \fi
  \ifdim\dimexpr\ht\tocbox+\dp\tocbox\relax>\toccap
    \setbox\tocpart\vsplit\tocbox to \toccap
    \edef\toc@last{\splitbotmark}%
    \setbox\toctmp\copy\tocbox
    \setbox\toctmp\vsplit\toctmp to 0pt
    \edef\toc@first{\splitfirstmark}%
    \setbox\toctmp\box\voidb@x
    \box\tocpart
    \columnbreak
    \advance\toccol\@ne
    \ifx\toc@first\toc@last
      \ifx\toc@last\@empty\else
        \setbox\tocbox\vbox{%
          \tocbarbox{\csname toc@name@\toc@last\endcsname\ (cont.)}%
          \unvbox\tocbox}%
      \fi
    \fi
    \expandafter\toc@place
  \else
    \box\tocbox
  \fi
}
