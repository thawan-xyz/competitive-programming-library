\section{Matching}

\textbf{Notação.} $M$ é um matching máximo, $|M|$ é o seu tamanho e $|V|$ é o número de vértices. Em bipartido,
$V = L \cup R$. Um vértice é \textbf{livre} se nenhuma aresta de $M$ toca nele.

\subsection*{Em bipartidos}

\begin{itemize}
  \item \textbf{König:} cobertura mínima de vértices (MVC), o menor conjunto de vértices que toca toda aresta,
        tem $|\text{MVC}| = |M|$.
  \item \textbf{Conjunto independente máximo (MIS):} o complemento de uma cobertura, $|\text{MIS}| = |V| - |M|$.
  \item \textbf{Cobertura mínima de arestas} (sem vértices isolados): $|V| - |M|$. Pegue o matching e adicione uma
        aresta qualquer de cada vértice livre.
  \item \textbf{Hall:} existe matching que casa todo $L$ $\iff$ $|N(X)| \ge |X|$ para todo $X \subseteq L$, onde
        $N(X)$ são os vizinhos de $X$. Em geral, $|M| = |L| - \max_{X \subseteq L} (|X| - |N(X)|)$, com
        $X = \emptyset$ valendo $0$.
  \item \textbf{Pesos nos vértices:} ligue $S \to u$ com $w_u$, $u \to v$ com $\infty$ e $v \to T$ com $w_v$
        ($u \in L$, $v \in R$). A MVC de peso mínimo é o corte mínimo, e o MIS de peso máximo é $\sum w - \text{corte}$.
\end{itemize}

\subsection*{Recuperar a resposta}

Com fluxo ($S \to L \to R \to T$, capacidade $1$), seja $Z$ o que a BFS de $S$ alcança no residual. Sem fluxo,
oriente as arestas fora de $M$ de $L$ para $R$ e as de $M$ de $R$ para $L$, e seja $Z$ o que se alcança a partir
dos vértices livres de $L$.
\begin{itemize}
  \item \textbf{Matching:} arestas $L \to R$ com fluxo $1$.
  \item \textbf{MVC:} $(L \setminus Z) \cup (R \cap Z)$.
  \item \textbf{MIS:} o complemento, $(L \cap Z) \cup (R \setminus Z)$.
\end{itemize}

\subsection*{Cobertura de DAG por caminhos}

Divida cada vértice $u$ em $u_L$ e $u_R$ e, para cada aresta $u \to v$, ligue $u_L \to v_R$. Cada aresta do
matching junta dois vértices no mesmo caminho.
\begin{itemize}
  \item \textbf{Sem repetir vértices:} o mínimo de caminhos é $|V| - |M|$.
  \item \textbf{Repetindo vértices:} faça o mesmo no fecho transitivo ($u \to v$ se $u$ alcança $v$). O mínimo
        $|V| - |M|$ é igual à maior anticadeia, o maior conjunto em que ninguém alcança ninguém (Dilworth).
  \item \textbf{Exemplo:} $a \to c$, $b \to c$, $c \to d$, $c \to e$ precisa de $3$ caminhos sem repetir, mas de
        $2$ repetindo ($acd$ e $bce$).
  \item \textbf{Maior anticadeia:} os $v$ com $v_L$ e $v_R$ fora da MVC do grafo dividido (no fecho).
  \item \textbf{Mirsky:} o menor número de anticadeias que cobre todos os vértices é o tamanho da maior cadeia
        (o caminho mais longo, contando vértices).
  \item \textbf{Caminhos:} siga as arestas $u_L \to v_R$ do matching; $u$ termina um caminho se $u_L$ está livre.
\end{itemize}

\subsection*{Em qualquer grafo}

\begin{itemize}
  \item \textbf{Gallai:} $|\text{MIS}| + |\text{MVC}| = |V|$ e, sem vértices isolados,
        $|M| + |\text{cobertura mínima de arestas}| = |V|$.
  \item König só vale em bipartido: no triângulo, $|\text{MVC}| = 2$ e $|M| = 1$.
\end{itemize}

\subsection*{Modelagens}

\begin{itemize}
  \item \textbf{Torres:} linhas de um lado, colunas do outro e uma aresta para cada casa $(i, j)$ permitida. O
        máximo de torres sem ataque é $|M|$, e o mínimo de linhas e colunas que cobrem todas as casas também (König).
\end{itemize}
